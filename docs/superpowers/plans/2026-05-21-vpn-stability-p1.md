* B1 — Block XMPP ports (5222/5223) at Xray routing layer — handled in `cmd/api/xray_sync.go` by appending a port-block rule before the default routing rule. Triggers `RequestXraySync()`.

    - reject(all, udp/0)
    # Common BitTorrent ports
    - reject(all, tcp/6881-6889)
    - reject(all, udp/6881-6889)
    - reject(all, tcp/51413)
    - reject(all, udp/51413)
    # Everything else — direct via tunnel
    - direct(all)

-- Per-user dial-rate observations sampled by dial_rate_monitor goroutine.
-- One row per (user_uuid, sample_window_start). 24h retention via periodic cleanup.
CREATE TABLE IF NOT EXISTS dial_rate_log (
    id BIGSERIAL PRIMARY KEY,
    user_uuid TEXT NOT NULL,
    window_start TIMESTAMPTZ NOT NULL,
    dials_in_window INTEGER NOT NULL,
    bytes_uplink BIGINT NOT NULL DEFAULT 0,
    bytes_downlink BIGINT NOT NULL DEFAULT 0,
    threshold_exceeded BOOLEAN NOT NULL DEFAULT false,
    UNIQUE (user_uuid, window_start)
);
CREATE INDEX IF NOT EXISTS idx_dial_rate_log_user_time
    ON dial_rate_log(user_uuid, window_start DESC);
CREATE INDEX IF NOT EXISTS idx_dial_rate_log_exceeded
    ON dial_rate_log(window_start DESC) WHERE threshold_exceeded = true;

-- Default threshold: 100 dials/min → ~1.66/sec sustained.
-- Median observed is ~10-15 dials/5min = 2-3/min. Top abusers hit 466 dials/5min = ~93/min.
INSERT INTO app_settings (key, value) VALUES
    ('dial_rate_threshold_per_min', '100')
ON CONFLICT (key) DO NOTHING;

```go
package biz

import (
	"strconv"

	"github.com/heyearth/backend/internal/db"
)

// DialRateSample is one observation window for a single user.
type DialRateSample struct {
	UserUUID       string
	WindowStartISO string // RFC3339
	Dials          int    // estimated dials in the window
	UpBytes        int64
	DownBytes      int64
}

// PersistDialRateSamples writes a batch of samples atomically.
// threshold is read from app_settings, may be tuned at runtime.
func PersistDialRateSamples(samples []DialRateSample, thresholdPerMin int) error {
	if len(samples) == 0 {
		return nil
	}
	tx, err := db.DB.Begin()
	if err != nil {
		return err
	}
	defer tx.Rollback()
	stmt, err := tx.Prepare(`
		INSERT INTO dial_rate_log
		    (user_uuid, window_start, dials_in_window, bytes_uplink, bytes_downlink, threshold_exceeded)
		VALUES ($1, $2::timestamptz, $3, $4, $5, $6)
		ON CONFLICT (user_uuid, window_start) DO NOTHING
	`)
	if err != nil {
		return err
	}
	defer stmt.Close()
	for _, s := range samples {
		exceeded := s.Dials > thresholdPerMin
		if _, err := stmt.Exec(s.UserUUID, s.WindowStartISO, s.Dials, s.UpBytes, s.DownBytes, exceeded); err != nil {
			return err
		}
	}
	return tx.Commit()
}

// GetDialRateThreshold returns the per-minute dial threshold from settings.
// Defaults to 100 if missing or malformed.
func GetDialRateThreshold() int {
	v := GetSettingOrDefault("dial_rate_threshold_per_min", "100")
	n, err := strconv.Atoi(v)
	if err != nil || n <= 0 {
		return 100
	}
	return n
}

// CleanupDialRateLog deletes samples older than 24 hours.
// Called once per hour from dial_rate_monitor.
func CleanupDialRateLog() (int64, error) {
	res, err := db.DB.Exec(`DELETE FROM dial_rate_log WHERE window_start < NOW() - INTERVAL '24 hours'`)
	if err != nil {
		return 0, err
	}
	return res.RowsAffected()
}
```

```go
package main

import (
	"context"
	"log"
	"strings"
	"time"

	"github.com/heyearth/backend/internal/biz"
	"github.com/heyearth/backend/internal/pb/xrayapi"
	"google.golang.org/grpc"
	"google.golang.org/grpc/credentials/insecure"
)

const (
	dialRateWindow  = 60 * time.Second
	dialRateXrayURL = "127.0.0.1:10085"
)

// StartDialRateMonitor запускает фоновый мониторинг dial-rate per user.
// Раз в 60 сек:
//   1. Запрашивает Xray Stats для всех user>>>{email}>>>traffic>>>uplink
//      (uplink monotonic counter — дельта между сэмплами = байты за окно)
//   2. Дельта uplink_bytes / N_USER_BYTES_PER_DIAL ≈ количество dial'ов
//      (груба, для наблюдения достаточно)
//   3. Пишет в dial_rate_log
//   4. Если делает >threshold/min — логирует WARN
//
// Threshold живёт в app_settings.dial_rate_threshold_per_min (по умолчанию 100).
// Раз в час чистит старые записи (>24h).
func StartDialRateMonitor() {
	go func() {
		time.Sleep(45 * time.Second) // дать Xray и БД подняться
		log.Println("[dial-rate-monitor] запущен (окно 60 сек)")

		prevSnapshot := map[string]int64{} // user_uuid → last uplink_bytes
		lastCleanup := time.Now()

		for {
			snapshot, err := fetchUserUplinkBytes()
			if err != nil {
				log.Printf("[dial-rate-monitor] Ошибка опроса Xray stats: %v", err)
				time.Sleep(dialRateWindow)
				continue
			}

			windowStart := time.Now().Truncate(time.Minute)
			windowStartISO := windowStart.Format(time.RFC3339)
			threshold := biz.GetDialRateThreshold()

			samples := make([]biz.DialRateSample, 0, len(snapshot))
			for uuid, cur := range snapshot {
				prev, ok := prevSnapshot[uuid]
				if !ok {
					// первый сэмпл — пропускаем чтобы не насчитать "весь uplink за uptime"
					continue
				}
				deltaBytes := cur - prev
				if deltaBytes < 0 {
					deltaBytes = 0 // counter wrap или reset Xray
				}
				// Примерно: 1 dial ≈ 50 KB трафика (TLS handshake + small request).
				// Это занижает для тяжёлых dial'ов (видео) и завышает для лёгких (DNS),
				// но для выявления abuser'ов достаточно: разница на порядок.
				estimatedDials := int(deltaBytes / 50_000)
				if estimatedDials > threshold {
					log.Printf("[dial-rate-monitor] WARN user=%s dials≈%d/min (threshold=%d) bytes=%dKB",
						uuid, estimatedDials, threshold, deltaBytes/1024)
				}
				samples = append(samples, biz.DialRateSample{
					UserUUID:       uuid,
					WindowStartISO: windowStartISO,
					Dials:          estimatedDials,
					UpBytes:        deltaBytes,
					DownBytes:      0, // отдельный counter, если нужно — добавить fetch
				})
			}

			if err := biz.PersistDialRateSamples(samples, threshold); err != nil {
				log.Printf("[dial-rate-monitor] Ошибка сохранения: %v", err)
			}

			prevSnapshot = snapshot

			// раз в час чистим старые сэмплы
			if time.Since(lastCleanup) > time.Hour {
				if n, err := biz.CleanupDialRateLog(); err == nil && n > 0 {
					log.Printf("[dial-rate-monitor] Очистка: %d старых записей удалено", n)
				}
				lastCleanup = time.Now()
			}

			time.Sleep(dialRateWindow)
		}
	}()
}

// fetchUserUplinkBytes опрашивает Xray StatsService и возвращает map user_uuid → uplink bytes.
func fetchUserUplinkBytes() (map[string]int64, error) {
	conn, err := grpc.Dial(dialRateXrayURL, grpc.WithTransportCredentials(insecure.NewCredentials()))
	if err != nil {
		return nil, err
	}
	defer conn.Close()
	client := xrayapi.NewStatsServiceClient(conn)
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	resp, err := client.QueryStats(ctx, &xrayapi.QueryStatsRequest{
		Pattern: "user>>>",
		Reset_:  false,
	})
	if err != nil {
		return nil, err
	}
	out := make(map[string]int64, len(resp.Stat))
	for _, s := range resp.Stat {
		// Name like: user>>>{email}>>>traffic>>>uplink
		if !strings.HasSuffix(s.Name, ">>>uplink") {
			continue
		}
		parts := strings.Split(s.Name, ">>>")
		if len(parts) < 4 {
			continue
		}
		email := parts[1]
		out[email] = s.Value
	}
	return out, nil
}
```
